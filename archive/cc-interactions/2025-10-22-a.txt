     3.1 Create Python Script to Parse Report

     # scripts/29_parse_accommodation_report.py

     """
     Parse accommodation_rationalization_report.md and generate tag assignments
     for appending to poly_hierarchy_additions.csv
     """

     import re
     from pathlib import Path

     def parse_hierarchy_path(path_str):
         """Parse pipe-separated hierarchy paths into individual tags."""
         paths = [p.strip() for p in path_str.split('|')]
         tags = []
         for path in paths:
             # Extract leaf tag (last element after final >)
             parts = [p.strip() for p in path.split('>')]
             if parts:
                 leaf_tag = parts[-1]
                 tags.append((leaf_tag, path))
         return tags

     def parse_report(report_path):
         """Parse report and extract tag assignments."""
         with open(report_path, 'r') as f:
             content = f.read()
         
         # Regex to extract each item block
         pattern = r'## (\d+)\. (.+?)\n.*?- \*\*Date:\*\* (.+?)\n.*?- \*\*Hierarchy Path:\*\* (.+?)\n'
         matches = re.findall(pattern, content, re.DOTALL)
         
         assignments = []
         for num, title, date, hierarchy_path in matches:
             if hierarchy_path == 'TBD':
                 continue  # Skip items needing manual review
             
             tags = parse_hierarchy_path(hierarchy_path)
             for tag, full_path in tags:
                 assignments.append({
                     'item_num': num,
                     'item_title': title,
                     'item_date': date,
                     'tag': tag,
                     'full_path': full_path
                 })
         
         return assignments

     # Run and output CSV rows
     assignments = parse_report('reports/accommodation_rationalization_report.md')
     print(f"Generated {len(assignments)} tag assignments for {len(set(a['item_num'] for a in assignments))} items")
